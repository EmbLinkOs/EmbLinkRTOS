if(CONFIG_EMB_CRASH_RECORD)
  # The build identity (A72): SHA-1 over the commit, whether tracked files differ from it, and
  # the configuration hash, fixed at configure time.
  execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${EMB_BASE}"
                  OUTPUT_VARIABLE _emb_commit OUTPUT_STRIP_TRAILING_WHITESPACE
                  RESULT_VARIABLE _emb_git ERROR_QUIET)
  if(NOT _emb_git EQUAL 0)
    set(_emb_commit "unknown")
  endif()
  execute_process(COMMAND git status --porcelain --untracked-files=no
                  WORKING_DIRECTORY "${EMB_BASE}" OUTPUT_VARIABLE _emb_dirty ERROR_QUIET)
  if(_emb_dirty)
    set(_emb_commit "${_emb_commit}+dirty")
  endif()
  string(SHA1 _emb_build_id "emb-build-id:${_emb_commit}:${EMB_CONFIG_HASH}")
  string(REGEX REPLACE "(..)" "0x\\1u, " _emb_bytes "${_emb_build_id}")
  set(EMB_BUILD_ID "${_emb_build_id}" CACHE INTERNAL "the image's build identity (A72)")
  file(GENERATE OUTPUT "${EMB_GENERATED_DIR}/build_id.c" CONTENT
       "/* Generated by kernel/CMakeLists.txt: ${_emb_commit}, configuration ${EMB_CONFIG_HASH}. */\n#include <embk/crash.h>\n\nEMB_FLASH_CONST uint8_t emb_build_id[EMB_CRASH_BUILD_ID_SIZE] = {${_emb_bytes}};\n")
  list(APPEND EMB_KERNEL_SOURCES fault/crash.c "${EMB_GENERATED_DIR}/build_id.c")
endif()
